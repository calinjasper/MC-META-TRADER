Placeholder_Updates(_Syntax = '', float _Strike = na, int _Qty = na, string _OPT_Type = na, float _SL_Price = na, float _TP_Price = na) =>
    _Return = _Syntax
    if not na(_Strike)
        _Return := str.replace_all(_Return, '{{STRIKE}}', str.tostring(int(_Strike)))
        _Return
    if not na(_Qty)
        _Return := str.replace_all(_Return, '{{QTY}}', str.tostring(_Qty))
        _Return
    if not na(_OPT_Type)
        _Return := str.replace_all(_Return, '{{OPT_TYPE}}', _OPT_Type)
        _Return
    if not na(_SL_Price)
        _Return := str.replace_all(_Return, '{{SL_PRICE}}', str.tostring(_SL_Price))
        _Return
    if not na(_TP_Price)
        _Return := str.replace_all(_Return, '{{TP_PRICE}}', str.tostring(_TP_Price))
        _Return
    _Return

Tooltip_String_Source = "Last input can  Accept Price Placholder Such as open, high, low, close, hl2, hlc3, ohlc4, hlcc4  \n&& as well as float also \n\nso only number or Placholder from above can be Accept here else it create error"
enum Conditions_Enum
    Greater_Than   = "Greater_Than"
    Lower_Than     = "Lower_Than"
    Equals         = "Equals"
    Crossover      = "Crossover"
    Crossunder     = "Crossunder"
    Any_Cross      = "Any_Cross"


type Conditions_Template
    string    ID     // LE, LX, SE, SX
    bool      Want 
    string    Condition
    float     Source_A
    float     Source_B
    bool      isCheck

method Check_Condition( Conditions_Template _CD )=>
    Condition__Greater_Than =                _CD.Source_A >  _CD.Source_B
    Condition__Lower_Than   =                _CD.Source_A <  _CD.Source_B
    Condition__Equals       =                _CD.Source_A == _CD.Source_B
    Condition__Crossover    = ta.crossover  (_CD.Source_A ,  _CD.Source_B)
    Condition__Crossunder   = ta.crossunder (_CD.Source_A ,  _CD.Source_B)
    Condition__Any_Cross    = ta.cross      (_CD.Source_A ,  _CD.Source_B)
    if _CD.Want
        _CD.isCheck := _CD.Condition==str.tostring( Conditions_Enum.Greater_Than ) ? Condition__Greater_Than 
                     : _CD.Condition==str.tostring( Conditions_Enum.Lower_Than   ) ? Condition__Lower_Than   
                     : _CD.Condition==str.tostring( Conditions_Enum.Equals       ) ? Condition__Equals       
                     : _CD.Condition==str.tostring( Conditions_Enum.Crossover    ) ? Condition__Crossover    
                     : _CD.Condition==str.tostring( Conditions_Enum.Crossunder   ) ? Condition__Crossunder   
                     : _CD.Condition==str.tostring( Conditions_Enum.Any_Cross    ) ? Condition__Any_Cross    
                     : true
    else
        _CD.isCheck := (_CD.ID=="LE" or _CD.ID=="SE" ? true : false)
        


Get_Source_From_String(string _String)=>
    _Return      = _String=='open'    ? open     
                 : _String=='high'    ? high     
                 : _String=='low'     ? low      
                 : _String=='close'   ? close    
                 : _String=='hl2'     ? hl2      
                 : _String=='hlc3'    ? hlc3     
                 : _String=='ohlc4'   ? ohlc4    
                 : _String=='hlcc4'   ? hlcc4    
                 : str.tonumber(_String)
    _Return


Get_Inputs(_ID, bool _Want, float _Source_A, string _Condition, float _Source_B=na,  string _Source_B_String=na )=>
    _Return = Conditions_Template.new(
                 ID        = _ID,
                 Want      = _Want,
                 Condition = _Condition,
                 Source_A  = _Source_A
                 )
    if na (_Source_B_String)
        _Return.Source_B := _Source_B
    else
        _Return.Source_B := Get_Source_From_String(_Source_B_String)
    _Return.Check_Condition()
    _Return

var Buy_TP2_Price            = float(na)
var Sell_TP2_Price           = float(na)

    if strategy.position_size < 0
        if low  < Sell_Trail_Start_Price
            New_Sell_SL_Price = math.min(low  + SL_Trailing_Points, Sell_SL_Price)
            if New_Sell_SL_Price != Sell_SL_Price
                Sell_SL_Price := New_Sell_SL_Price
                Sell_Trail_SL_Updated := true

// TP1 and TP2 Exit Logic

Sell_TP1_Hit_Condition  = (close <= Sell_TP1_Price or low <= Sell_TP1_Price) and strategy.position_size<0 and not Sell_TP1_Hit

// TP2: Only check if TP1 was already hit on a previous bar AND price actually reaches TP2 (must be > TP1 for Buy, < TP1 for Sell)
// IMPORTANT: TP2 is NOT included in Buy_Exit/Sell_Exit - it's handled separately to prevent same-bar execution
Buy_TP2_Hit_Condition   = strategy.position_size>0 and Buy_TP1_Hit_Previous and Buy_TP2_Price > Buy_TP1_Price and (TP_Execute_Type == 'On Close' ? close >= Buy_TP2_Price : high >= Buy_TP2_Price)
Sell_TP2_Hit_Condition  = strategy.position_size<0 and Sell_TP1_Hit_Previous and Sell_TP2_Price < Sell_TP1_Price and (TP_Execute_Type == 'On Close' ? close <= Sell_TP2_Price : low <= Sell_TP2_Price)

Buy_Exit_3__TP1_Hit      = false  // TP1 is handled separately, not through Buy_Exit
Sell_Exit_3__TP1_Hit     = false  // TP1 is handled separately, not through Sell_Exit
Buy_Exit_4__TP2_Hit      = false  // TP2 is handled separately, not through Buy_Exit
Sell_Exit_4__TP2_Hit     = false  // TP2 is handled separately, not through Sell_Exit

Buy_Exit_5__SL_Hit       =  close <  Buy_SL_Price and strategy.position_size>0 
Sell_Exit_5__SL_Hit      =  close > Sell_SL_Price and strategy.position_size<0 

Buy_Exit    = Buy_Exit_1   or Buy_Exit_2   or Buy_Exit_5__SL_Hit  
Sell_Exit   = Sell_Exit_1  or Sell_Exit_2  or Sell_Exit_5__SL_Hit 





if Buy_Time
    Buy_EP_Price       := close 
    Buy_TP1_Hit        := false
    Buy_TP1_Hit_Previous := false
    Buy_SL_Moved_To_BreakEven := false
    
    if Want_SL
        if      SL_Type=="Percent"
            Buy_SL_Price   := Buy_EP_Price - (Buy_EP_Price*0.01*SL_Points_Percent)     
        else if SL_Type=="Points"
            Buy_SL_Price   := Buy_EP_Price - SL_Points_Percent
        else if SL_Type=="Swing"
            Buy_SL_Price   := SL_Low
    
    if Want_TP
        // Calculate TP1 and TP2
        if      TP_Type=="Percent"
            Buy_TP1_Price   := Buy_EP_Price + (Buy_EP_Price*0.01*TP1_Points_Percent)
            Buy_TP2_Price   := Buy_EP_Price + (Buy_EP_Price*0.01*TP2_Points_Percent)
        else if TP_Type=="Points"
            Buy_TP1_Price   := Buy_EP_Price + TP1_Points_Percent
            Buy_TP2_Price   := Buy_EP_Price + TP2_Points_Percent
        else if TP_Type=="RR"
            SL_Points       = math.abs(Buy_EP_Price - Buy_SL_Price)
            Buy_TP1_Price   := Buy_EP_Price + TP1_Points_Percent*SL_Points
            Buy_TP2_Price   := Buy_EP_Price + TP2_Points_Percent*SL_Points
    
    if Want_Trailing_SL
        Buy_Trail_Start_Price   := Buy_EP_Price + Trail_Start_Points
    
    // Fire Buy Entry Alert
    if iAlgo_Buy_Entry_Syntax != ''
        Alert_Message = Algo_Buy_Entry_Syntax('OFF', Price = Buy_EP_Price, Strike = Buy_EP_Price, SL_Price = Buy_SL_Price, TP_Price = Buy_TP1_Price)
        alert(Alert_Message, alert.freq_once_per_bar_close)


if Sell_Time
    Sell_EP_Price       := close 
    Sell_TP1_Hit        := false
    Sell_TP1_Hit_Previous := false
    Sell_SL_Moved_To_BreakEven := false
    
    if Want_SL

            Sell_SL_Price   := Sell_EP_Price + (Sell_EP_Price*0.01*SL_Points_Percent)

        else if SL_Type=="Swing"
            Sell_SL_Price   := SL_High

    if Want_TP

        if      TP_Type=="Percent"
            Sell_TP1_Price   := Sell_EP_Price - (Sell_EP_Price*0.01*TP1_Points_Percent)
            Sell_TP2_Price   := Sell_EP_Price - (Sell_EP_Price*0.01*TP2_Points_Percent)
        else if TP_Type=="Points"
            Sell_TP1_Price   := Sell_EP_Price - TP1_Points_Percent
            Sell_TP2_Price   := Sell_EP_Price - TP2_Points_Percent
        else if TP_Type=="RR"
            SL_Points       = math.abs(Sell_EP_Price - Sell_SL_Price)
            Sell_TP1_Price   := Sell_EP_Price - TP1_Points_Percent*SL_Points
            Sell_TP2_Price   := Sell_EP_Price - TP2_Points_Percent*SL_Points

    if Want_Trailing_SL
        Sell_Trail_Start_Price   := Sell_EP_Price - Trail_Start_Points
